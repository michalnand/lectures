\begin{frame}{\bf Action sampling}

  \centering{\includegraphics[scale=0.3]{../rl_images/a2c_logits_actions.png}}

  \begin{enumerate}
    \item logits : $$l(s) = f_\theta(s)$$
    \item probs : $$\pi_{\theta}(a|s)=softmax(l(s))$$
    \item action sampling : $$a \sim \pi_{\theta}(.|s)$$
  \end{enumerate}

\end{frame}

\begin{frame}{\bf Actor training}

  \centering{\includegraphics[scale=0.3]{../rl_images/a2c_logits_actions_comparison.png}}

  \begin{enumerate}
    \item advantages (simplified)  $$A(s, a) = reward - V(s)$$
    \item policy loss              $$\mathcal{L}_{\theta}(s, a) = -A(s, a)log \pi_{\theta}(a|s)$$
  \end{enumerate}

\end{frame}


\begin{frame}{\bf Critic training}

  \centering{\includegraphics[scale=0.4]{../rl_images/a2c_critic_value_comparison.png}}

  \begin{enumerate}
    \item critic loss     $$\mathcal{L}_{\Phi}(s)= (V(s) - r)^2$$
  \end{enumerate}

\end{frame}
